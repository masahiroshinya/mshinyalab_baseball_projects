def get_video_info(path):
    cap = cv2.VideoCapture(path)
    if not cap.isOpened():
        raise FileNotFoundError(f"動画ファイルを開けません: {path}")
    info = {
        "path":       path,
        "fps":        cap.get(cv2.CAP_PROP_FPS),
        "width":      int(cap.get(cv2.CAP_PROP_FRAME_WIDTH)),
        "height":     int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT)),
        "n_frames":   int(cap.get(cv2.CAP_PROP_FRAME_COUNT)),
    }
    info["duration_sec"] = info["n_frames"] / info["fps"] if info["fps"] > 0 else 0
    cap.release()
    return info

def compute_per_frame_stereo_error(obj_list, img1_list, img2_list,
                                   mtx1, dist1, mtx2, dist2, R, T):
    """
    各フレームのステレオ再投影誤差（px）をリストで返す。

    アルゴリズム：
      1. Camera1 の検出点から solvePnP でボードの姿勢（rvec1, tvec1）を推定
      2. ステレオの R/T を使って Camera2 でのボード姿勢を計算
           R2 = R @ R1_mat,   t2 = R @ tvec1 + T
      3. Camera1 / Camera2 それぞれで projectPoints → 実測値との差の RMS を計算
      4. 2台の平均誤差を返す
    """
    errors = []
    for obj_pts, img1_pts, img2_pts in zip(obj_list, img1_list, img2_list):
        try:
            ok, rvec1, tvec1 = cv2.solvePnP(
                obj_pts.reshape(-1, 3), img1_pts, mtx1, dist1,
                flags=cv2.SOLVEPNP_ITERATIVE)
            if not ok:
                errors.append(np.inf); continue
            proj1, _ = cv2.projectPoints(obj_pts.reshape(-1, 3), rvec1, tvec1, mtx1, dist1)
            err1 = np.sqrt(np.mean((img1_pts.reshape(-1, 2) - proj1.reshape(-1, 2))**2))
            R1_mat, _ = cv2.Rodrigues(rvec1)
            rvec2, _  = cv2.Rodrigues(R @ R1_mat)
            t2        = R @ tvec1 + T
            proj2, _ = cv2.projectPoints(obj_pts.reshape(-1, 3), rvec2, t2, mtx2, dist2)
            err2 = np.sqrt(np.mean((img2_pts.reshape(-1, 2) - proj2.reshape(-1, 2))**2))
            errors.append((err1 + err2) / 2.0)
        except Exception:
            errors.append(np.inf)
    return errors


def stereo_calibrate_robust(objpoints_s, imgpts1_s, imgpts2_s,
                             mtx1, dist1, mtx2, dist2, image_size,
                             rms_target=1.5, min_frames=6, max_remove=5):
    """
    外れ値フレームを逐次除去しながらステレオキャリブレーションを実行する。

    Parameters:
        rms_target  : この値以下になったら終了（px）
        min_frames  : これ以下になったら強制終了
        max_remove  : 除去するフレームの上限枚数

    Returns:
        (ret_s, mtx1, dist1, mtx2, dist2, R, T, E, F, n_used)
    """
    obj_list  = list(objpoints_s)
    img1_list = list(imgpts1_s)
    img2_list = list(imgpts2_s)
    criteria  = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 100, 1e-6)
    print(f"  初期ペア数: {len(obj_list)}")
    for iteration in range(max_remove + 1):
        ret_s, m1, d1, m2, d2, R, T, E, F = cv2.stereoCalibrate(
            obj_list, img1_list, img2_list,
            mtx1, dist1, mtx2, dist2, image_size,
            criteria=criteria, flags=cv2.CALIB_FIX_INTRINSIC)
        mark = "✅" if ret_s <= rms_target else "⚠"
        print(f"  [{iteration}] RMS={ret_s:.4f}px  ペア数={len(obj_list)}  {mark}")
        if ret_s <= rms_target:
            print("  目標RMS達成"); break
        if len(obj_list) <= min_frames or iteration == max_remove:
            print("  ⚠ 目標未達。キャリブ画像の多様性不足の可能性があります。")
            print("    推奨: ボードを傾ける・端に寄せる・距離を変えるなど 20枚以上で再撮影")
            break
        per_errors = compute_per_frame_stereo_error(
            obj_list, img1_list, img2_list, mtx1, dist1, mtx2, dist2, R, T)
        worst = int(np.argmax(per_errors))
        print(f"    → ペア {worst} を除外 (誤差: {per_errors[worst]:.4f}px)")
        obj_list.pop(worst); img1_list.pop(worst); img2_list.pop(worst)
    return ret_s, m1, d1, m2, d2, R, T, E, F, len(obj_list)


print("✅ セットアップ完了")

def pick_high_energy_peaks(audio, sr, n_peaks=5, min_dist_sec=0.8, energy_window_sec=0.1):
    energy = np.convolve(audio ** 2,
                         np.ones(int(sr * energy_window_sec), dtype=np.float32) / max(1, int(sr * energy_window_sec)),
                         mode='same')
    peak_idxs, _ = signal.find_peaks(energy, distance=int(min_dist_sec * sr))
    if len(peak_idxs) < n_peaks:
        sorted_idxs = np.argsort(energy)[::-1]
        for idx in sorted_idxs:
            if idx in peak_idxs:
                continue
            peak_idxs = np.append(peak_idxs, idx)
            if len(peak_idxs) >= n_peaks:
                break
    return np.array(sorted(peak_idxs[:n_peaks]))


def compute_local_alignment_quality(ref_signal, target_signal, offset_samp, sr, n_peaks=5, window_sec=0.5):
    peak_idxs = pick_high_energy_peaks(ref_signal, sr, n_peaks=n_peaks)
    if len(peak_idxs) == 0:
        return []
    window_len = int(window_sec * sr)
    local_sims = []
    for idx in peak_idxs:
        start_ref = max(0, idx - window_len // 2)
        end_ref = min(len(ref_signal), start_ref + window_len)
        start_ref = max(0, end_ref - window_len)
        start_target = start_ref + offset_samp
        end_target = start_target + (end_ref - start_ref)
        if start_target < 0 or end_target > len(target_signal):
            continue
        ref_seg = ref_signal[start_ref:end_ref]
        tgt_seg = target_signal[start_target:end_target]
        ref_seg = (ref_seg - np.mean(ref_seg)) / (np.std(ref_seg) + 1e-8)
        tgt_seg = (tgt_seg - np.mean(tgt_seg)) / (np.std(tgt_seg) + 1e-8)
        sim = float(np.dot(ref_seg, tgt_seg) / len(ref_seg))
        local_sims.append((idx, start_ref / sr, sim))
    return local_sims


def compute_audio_offset(audio_ref, audio_target, sr, fps):
    ref_norm    = (audio_ref    - np.mean(audio_ref))    / (np.std(audio_ref)    + 1e-8)
    target_norm = (audio_target - np.mean(audio_target)) / (np.std(audio_target) + 1e-8)
    USE_SEC = 60
    n_use = min(int(USE_SEC * sr), len(ref_norm), len(target_norm))
    print(f"  クロスコリレーション計算中（先頭 {n_use/sr:.0f} 秒分）...")
    correlation  = signal.correlate(ref_norm[:n_use], target_norm[:n_use], mode='full')
    lags_samples = signal.correlation_lags(n_use, n_use, mode='full')
    peak_idx     = np.argmax(correlation)
    offset_samp  = lags_samples[peak_idx]
    offset_sec   = offset_samp / sr
    offset_frames = int(round(offset_sec * fps))
    norm_peak    = correlation[peak_idx] / n_use
    local_sims   = compute_local_alignment_quality(ref_norm[:n_use], target_norm[:n_use], offset_samp, sr)
    return offset_frames, offset_sec, offset_samp, correlation, lags_samples, norm_peak, local_sims

print("=== 音声クロスコリレーション ===")

def compute_start_frames(offsets_frames):
    max_offset = max(offsets_frames)
    return [max_offset - off for off in offsets_frames]

start_frames  = compute_start_frames(offsets_frames)
common_frames = min(video_infos[i]["n_frames"] - start_frames[i] for i in range(nCamera))

print("=== クリップ開始フレームの決定 ===")

    print(f"  Camera {i+1}: 先頭 {start_frames[i]} frames ({start_frames[i]/base_fps:.3f} sec) をスキップ")
print(f"\n  共通フレーム数: {common_frames} frames = {common_frames/base_fps:.1f} sec")

def clip_and_save(video_paths, start_frames, common_frames, video_infos):
    """ffmpeg stream copy で高速クリップ（再エンコードなし）"""
    fps          = video_infos[0]["fps"]
    duration_sec = common_frames / fps
    output_paths = []
    for i, (path, sf) in enumerate(zip(video_paths, start_frames)):
        out_path  = f"cam{i+1}_synced.mp4"
        start_sec = sf / fps
        cmd = [
            "ffmpeg", "-y",
            "-ss", f"{start_sec:.6f}",   # -i の前に置く（高速シーク）
            "-i",  path,
            "-t",  f"{duration_sec:.6f}",
            "-c",  "copy",               # 再エンコードなし
            out_path,
        ]
        print(f"Camera {i+1}: クリップ中 (開始:{start_sec:.3f}s, 長さ:{duration_sec:.1f}s)...")
        result = subprocess.run(cmd, capture_output=True, text=True)
        if result.returncode != 0:
            raise RuntimeError(f"Camera {i+1} ffmpeg エラー:\n{result.stderr[-600:]}")
        size_mb = os.path.getsize(out_path) / (1024 * 1024)
        print(f"  ✅ {out_path} ({size_mb:.1f} MB)")
        output_paths.append(out_path)
    return output_paths

print("\n=== 映像クリップ・保存開始 ===")
output_paths = clip_and_save(video_paths, start_frames, common_frames, video_infos)

def calibrate_single_camera(corners_list, ids_list, img_size):
    all_obj_pts, all_img_pts = [], []
    for corners, ids in zip(corners_list, ids_list):
        obj_pts = np.array([board9_corners[idx[0]] for idx in ids],
                           dtype=np.float32).reshape(-1, 1, 3)
        all_obj_pts.append(obj_pts); all_img_pts.append(corners)
    fx_init = img_size[0] * 0.8
    K_init  = np.array([[fx_init, 0, img_size[0]/2],
                         [0, fx_init, img_size[1]/2],
                         [0, 0, 1]], dtype=np.float64)
    return cv2.calibrateCamera(
        all_obj_pts, all_img_pts, img_size, K_init, None,
        flags=cv2.CALIB_USE_INTRINSIC_GUESS | cv2.CALIB_FIX_K3)

def get_2d_points(img):
    kps = None
    results_pose = pose_model(img, verbose=False)
    if results_pose and results_pose[0].keypoints is not None:
        kp_data = results_pose[0].keypoints.xy.cpu().numpy()
        if len(kp_data) > 0:
            boxes = results_pose[0].boxes
            if boxes is not None and len(boxes) > 0:
                areas = [(b[2]-b[0])*(b[3]-b[1]) for b in boxes.xyxy.cpu().numpy()]
                best  = int(np.argmax(areas))
            else:
                best = 0
            kps = kp_data[best][:, :2]
    bat_center = None
    if DETECT_BAT and detect_model is not None:
        results_det = detect_model(img, verbose=False)
        best_bat, best_conf = None, 0
        if results_det and results_det[0].boxes is not None:
            for box in results_det[0].boxes:
                cls  = int(box.cls.item())
                conf = float(box.conf.item())
                if cls == BAT_CLASS_ID and conf > best_conf:
                    best_bat  = box.xyxy.cpu().numpy()[0]
                    best_conf = conf
        if best_bat is not None:
            bat_center = np.array([(best_bat[0]+best_bat[2])/2,
                                   (best_bat[1]+best_bat[3])/2])
    return kps, bat_center

P1_norm = np.hstack([np.eye(3), np.zeros((3, 1))])
P2_norm = np.hstack([R, T])

    ret2_v, img2 = cap2.read()
    if not (ret1_v and ret2_v): continue

    kps1, bat1 = get_2d_points(img1)
    kps2, bat2 = get_2d_points(img2)

    if kps1 is not None and kps2 is not None:
        pts1_n = cv2.undistortPoints(np.expand_dims(kps1, 1), mtx1, dist1)
        pts2_n = cv2.undistortPoints(np.expand_dims(kps2, 1), mtx2, dist2)
        pts4d  = cv2.triangulatePoints(P1_norm, P2_norm,
                                        pts1_n.reshape(-1, 2).T, pts2_n.reshape(-1, 2).T)
        pts3d  = (pts4d[:3] / pts4d[3]).T
        for i, name in enumerate(KEYPOINT_NAMES):
            row_data[f"{name}_X"] = round(pts3d[i, 0], 4)
            row_data[f"{name}_Y"] = round(pts3d[i, 1], 4)
            row_data[f"{name}_Z"] = round(pts3d[i, 2], 4)
    else:
        for name in KEYPOINT_NAMES:
            row_data[f"{name}_X"] = row_data[f"{name}_Y"] = row_data[f"{name}_Z"] = ""

    if DETECT_BAT and bat1 is not None and bat2 is not None:
        b1_n = cv2.undistortPoints(bat1.reshape(1, 1, 2), mtx1, dist1)
        b2_n = cv2.undistortPoints(bat2.reshape(1, 1, 2), mtx2, dist2)
        b4d  = cv2.triangulatePoints(P1_norm, P2_norm, b1_n.reshape(2, 1), b2_n.reshape(2, 1))
        b3d  = (b4d[:3] / b4d[3]).flatten()
        row_data["bat_X"] = round(b3d[0], 4)
        row_data["bat_Y"] = round(b3d[1], 4)
        row_data["bat_Z"] = round(b3d[2], 4)
    else:
        row_data["bat_X"] = row_data["bat_Y"] = row_data["bat_Z"] = ""

    results_3d.append(row_data)

cap1.release(); cap2.release()

if results_3d:
