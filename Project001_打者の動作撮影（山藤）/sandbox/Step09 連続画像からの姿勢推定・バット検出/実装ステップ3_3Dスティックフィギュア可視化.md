---

## アルゴリズム解説

### なぜ `clear_output(wait=True)` を使うか
Colabの`ipywidgets`とmatplotlibを組み合わせる場合、フレームごとに新しい`figure`を生成して`clear_output`で前のものを消す方式が最も安定する。`wait=True`にすると、次の描画が準備できてから表示を切り替えるため、画面のちらつきを防げる。

### なぜ軸スケールを全フレームで固定するか
フレームごとに軸の範囲を自動調整すると、骨格が小さく見えたり大きく見えたりして動きが把握しにくくなる。全フレームの座標から重心と範囲を事前計算して固定することで、スティックフィギュアが安定して見える。

### スケルトンの接続（COCOフォーマット 17点）
```
インデックス：  0=nose, 1=left_eye, 2=right_eye, 3=left_ear, 4=right_ear,
                5=left_shoulder, 6=right_shoulder, 7=left_elbow, 8=right_elbow,
                9=left_wrist, 10=right_wrist, 11=left_hip, 12=right_hip,
                13=left_knee, 14=right_knee, 15=left_ankle, 16=right_ankle
```

## セル 3：ライブラリ読み込みとデータ準備

```python
import cv2
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D
import ipywidgets as widgets
from IPython.display import display, clear_output

# ---- キーポイント定義 ----
KEYPOINT_NAMES = [
    "nose", "left_eye", "right_eye", "left_ear", "right_ear",
    "left_shoulder", "right_shoulder", "left_elbow", "right_elbow",
    "left_wrist", "right_wrist", "left_hip", "right_hip",
    "left_knee", "right_knee", "left_ankle", "right_ankle"
]

# ---- スケルトンの接続リスト（インデックスのペア） ----
SKELETON_CONNECTIONS = [
    (0, 1), (0, 2),          # 鼻 → 目
    (1, 3), (2, 4),          # 目 → 耳
    (5, 6),                  # 左肩 → 右肩
    (5, 7), (7, 9),          # 左腕
    (6, 8), (8, 10),         # 右腕
    (5, 11), (6, 12),        # 肩 → 腰
    (11, 12),                # 左腰 → 右腰
    (11, 13), (13, 15),      # 左脚
    (12, 14), (14, 16),      # 右脚
]

# ---- CSVデータの読み込み ----
df = pd.read_csv(CSV_3D_PATH)

# nose_X が有効な行だけを対象にする
valid_mask = pd.to_numeric(df["nose_X"], errors='coerce').notna()
df_valid = df[valid_mask].copy().reset_index(drop=True)
frame_numbers = df_valid["frame_number"].astype(int).tolist()

print(f"✅ 有効フレーム数: {len(frame_numbers)}")
print(f"   フレーム範囲: {frame_numbers[0]} 〜 {frame_numbers[-1]}")

# ---- 全フレームにわたる座標範囲の事前計算（軸スケール固定用） ----
all_x, all_y, all_z = [], [], []
for kp in KEYPOINT_NAMES:
    for col, lst in [(f"{kp}_X", all_x), (f"{kp}_Y", all_y), (f"{kp}_Z", all_z)]:
        if col in df_valid.columns:
            vals = pd.to_numeric(df_valid[col], errors='coerce').dropna()
            lst.extend(vals.tolist())

GLOBAL_MID_X = np.mean(all_x) if all_x else 0.0
GLOBAL_MID_Y = np.mean(all_y) if all_y else 0.0
GLOBAL_MID_Z = np.mean(all_z) if all_z else 0.0

print(f"   3D重心 → X:{GLOBAL_MID_X:.3f}, Y:{GLOBAL_MID_Y:.3f}, Z:{GLOBAL_MID_Z:.3f} [m]")

# ---- ビデオキャプチャの初期化 ----
cap1 = cv2.VideoCapture(VIDEO1_PATH)
cap2 = cv2.VideoCapture(VIDEO2_PATH)

def get_frame_image(cap, frame_no):
    """動画からRGB画像を取得する"""
    cap.set(cv2.CAP_PROP_POS_FRAMES, frame_no)
    ret, frame = cap.read()
    if ret:
        return cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
    return np.zeros((480, 640, 3), dtype=np.uint8)

def get_3d_keypoints(row):
    """DataFrameの1行から17点の3D座標リストを返す（無効はNone）"""
    pts = []
    for kp in KEYPOINT_NAMES:
        x = pd.to_numeric(row.get(f"{kp}_X", np.nan), errors='coerce')
        y = pd.to_numeric(row.get(f"{kp}_Y", np.nan), errors='coerce')
        z = pd.to_numeric(row.get(f"{kp}_Z", np.nan), errors='coerce')
        if np.isnan(x) or np.isnan(y) or np.isnan(z):
            pts.append(None)
        else:
            pts.append((float(x), float(y), float(z)))
    return pts

print("✅ データ準備完了")
```

## セル 4：インタラクティブ可視化（スライダー）

```python
# ============================================================
# 可視化メイン関数
# ============================================================
def update_viz(frame_idx):
    """スライダーの値（インデックス）が変わるたびに呼ばれる"""
    clear_output(wait=True)   # 前の描画を消す（wait=Trueでちらつき防止）
    frame_no = frame_numbers[frame_idx]

    # --- データ取得 ---
    row     = df_valid[df_valid["frame_number"] == frame_no].iloc[0]
    img1    = get_frame_image(cap1, frame_no)
    img2    = get_frame_image(cap2, frame_no)
    pts3d   = get_3d_keypoints(row)

    bat_xyz = None
    if "bat_X" in row.index and pd.notna(row["bat_X"]):
        bat_xyz = (float(row["bat_X"]), float(row["bat_Y"]), float(row["bat_Z"]))

    # --- 描画レイアウト ---
    fig = plt.figure(figsize=(18, 6))
    fig.patch.set_facecolor('#1a1a2e')

    # [左] Camera 1
    ax1 = fig.add_subplot(1, 3, 1)
    ax1.imshow(img1)
    ax1.set_title(f'📷 Camera 1  |  Frame {frame_no}',
                  color='white', fontsize=12, pad=8)
    ax1.axis('off')

    # [中] Camera 2
    ax2 = fig.add_subplot(1, 3, 2)
    ax2.imshow(img2)
    ax2.set_title(f'📷 Camera 2  |  Frame {frame_no}',
                  color='white', fontsize=12, pad=8)
    ax2.axis('off')

    # [右] 3D スティックフィギュア
    ax3 = fig.add_subplot(1, 3, 3, projection='3d')
    ax3.set_facecolor('#0d0d1a')

    # 骨格リンク（線）
    for (i, j) in SKELETON_CONNECTIONS:
        if pts3d[i] is not None and pts3d[j] is not None:
            ax3.plot(
                [pts3d[i][0], pts3d[j][0]],
                [pts3d[i][1], pts3d[j][1]],
                [pts3d[i][2], pts3d[j][2]],
                color='#00d4ff', linewidth=2.5, alpha=0.9
            )

    # 関節点（丸）
    for pt in pts3d:
        if pt is not None:
            ax3.scatter(*pt, color='#ff6b6b', s=40, zorder=5)

    # バット中心（星）
    if bat_xyz is not None:
        ax3.scatter(*bat_xyz, color='#ffd700', s=120,
                    marker='*', zorder=6, label='Bat center')
        ax3.legend(loc='upper right', fontsize=8,
                   labelcolor='white', facecolor='#1a1a2e', edgecolor='gray')

    # 軸スケール固定（全フレーム共通）
    ax3.set_xlim(GLOBAL_MID_X - AXIS_RANGE, GLOBAL_MID_X + AXIS_RANGE)
    ax3.set_ylim(GLOBAL_MID_Y - AXIS_RANGE, GLOBAL_MID_Y + AXIS_RANGE)
    ax3.set_zlim(GLOBAL_MID_Z - AXIS_RANGE, GLOBAL_MID_Z + AXIS_RANGE)
    ax3.set_xlabel('X (m)', color='gray', fontsize=9)
    ax3.set_ylabel('Y (m)', color='gray', fontsize=9)
    ax3.set_zlabel('Z (m)', color='gray', fontsize=9)
    ax3.tick_params(colors='gray', labelsize=7)
    ax3.set_title('🏃 3D Skeleton', color='white', fontsize=12, pad=8)
    ax3.view_init(elev=VIEW_ELEV, azim=VIEW_AZIM)

    plt.tight_layout(pad=0.5)
    display(fig)
    plt.close(fig)   # メモリ解放


# ============================================================
# スライダーUI
# ============================================================
slider = widgets.IntSlider(
    value=0,
    min=0,
    max=len(frame_numbers) - 1,
    step=1,
    description='フレーム:',
    continuous_update=False,   # マウスを離したときだけ更新（重い処理向け）
    layout=widgets.Layout(width='90%'),
    style={'description_width': '80px'}
)

frame_label = widgets.HTML(
    value=f'<b style="color:#00d4ff; font-size:14px;">Frame No: {frame_numbers[0]}</b>'
)

def on_slider_change(change):
    idx = change["new"]
    frame_label.value = (
        f'<b style="color:#00d4ff; font-size:14px;">'
        f'Frame No: {frame_numbers[idx]}</b>'
    )
    update_viz(idx)

slider.observe(on_slider_change, names='value')

# 初期描画
display(widgets.VBox([
    widgets.HBox([slider, frame_label]),
]))
update_viz(0)
```

## 表示調整のヒント

| パラメータ | 説明 | 推奨値 |
|-----------|------|--------|
| `AXIS_RANGE` | 3D軸の表示範囲 ±[m] | `1.0`〜`1.5` |
| `VIEW_ELEV` | 3D視点の仰角（上から見る角度） | `15`〜`30` |
| `VIEW_AZIM` | 3D視点の方位角（横から見る角度） | `-90`〜`-60` |
| `continuous_update=False` | スライダーをドラッグ中は更新しない（重い処理向け） | そのまま |

---

## 注意事項

- `batting_3d_tracking.csv` の `nose_X` が空のフレームはスキップされる（三角測量に失敗したフレーム）。
- `cap1`, `cap2` はセル3を再実行しないと初期化されないため、エラーが出た場合はセル3から再実行すること。
- 表示が重い場合は `AXIS_RANGE` の値を小さくするか、figsize を `(14, 5)` 程度に縮小する。
