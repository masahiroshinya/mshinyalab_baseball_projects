## 目標

`Software 3` などで抽出された **打撃区間の連続画像（複数枚）** に対して、YOLOv8（バット検出）と YOLOv8-pose（骨格検出）を一括で実行します。
各フレームから「バットを持っている打者」を自動で特定し、その17点の骨格座標とバットのバウンディングボックスを **1つのCSVファイルにまとめて保存** します。

## ステップ 3：一括処理スクリプトの実行

次のセルに以下のスクリプトをすべてコピー＆ペーストして実行します。
**注意:** 環境に合わせて、最初の `--- 設定 ---` ブロックにある `INPUT_DIR` などのパスを書き換えてください。

```python
import os
import glob
import cv2
import pandas as pd
import numpy as np
from ultralytics import YOLO
from google.colab.patches import cv2_imshow # 必要に応じて可視化用

# ============================================
# --- 設定 ---
# ============================================
# 解析したい画像が入っているフォルダのパス
INPUT_DIR = "/content/input_frames"    

# 結果画像（骨格やバットを描画した画像）を保存するフォルダ
OUTPUT_DIR = "/content/output_frames"  

# 座標データをまとめたCSVの出力先
OUTPUT_CSV = "/content/batting_2d_tracking.csv"

# モデルの指定（Colab T4なら m サイズが高精度かつ実用的）
POSE_MODEL_NAME = "yolov8m-pose.pt"
DETECT_MODEL_NAME = "yolov8m.pt"

BAT_CLASS_ID = 34  # COCOデータセットにおける 'baseball bat'

# COCO 17点キーポイント名
KEYPOINT_NAMES = [
    "nose", "left_eye", "right_eye", "left_ear", "right_ear",
    "left_shoulder", "right_shoulder", "left_elbow", "right_elbow",
    "left_wrist", "right_wrist", "left_hip", "right_hip",
    "left_knee", "right_knee", "left_ankle", "right_ankle"
]

# ============================================
# --- 初期化 ---
# ============================================
os.makedirs(OUTPUT_DIR, exist_ok=True)

print("モデルを読み込んでいます...")
pose_model = YOLO(POSE_MODEL_NAME)
detect_model = YOLO(DETECT_MODEL_NAME)

# 画像一覧の取得と名前順ソート
img_paths = sorted(glob.glob(os.path.join(INPUT_DIR, "*.[jp][pn]g"))) # .jpg, .pngに対応
print(f"見つかった画像数: {len(img_paths)}枚\n")

all_rows = []

# ============================================
# --- メインループ ---
# ============================================
for frame_idx, img_path in enumerate(img_paths):
    filename = os.path.basename(img_path)
    print(f"Processing {filename} ({frame_idx+1}/{len(img_paths)})...")
    
    img = cv2.imread(img_path)
    if img is None:
        continue
        
    # ----------------------------------------
    # 1. バットの検出 (YOLOv8m)
    # ----------------------------------------
    det_results = detect_model(img, verbose=False)
    bat_boxes = []
    
    for r in det_results:
        boxes = r.boxes
        for i in range(len(boxes)):
            cls_id = int(boxes.cls[i].item())
            if cls_id == BAT_CLASS_ID:
                x1, y1, x2, y2 = boxes.xyxy[i].tolist()
                conf = float(boxes.conf[i].item())
                bat_boxes.append([x1, y1, x2, y2, conf])
                
    # 複数検出された場合、最も信頼度の高いものを「バット」とみなす
    best_bat = None
    if bat_boxes:
        bat_boxes.sort(key=lambda x: x[4], reverse=True)
        best_bat = bat_boxes[0]
        
    # ----------------------------------------
    # 2. 姿勢（骨格）の検出 (YOLOv8m-pose) と 打者の特定
    # ----------------------------------------
    pose_results = pose_model(img, verbose=False)
    
    target_person_kps = None
    target_person_box = None
    
    for r in pose_results:
        if r.keypoints is None or len(r.keypoints.data) == 0:
            continue
            
        kps_data = r.keypoints.data.cpu().numpy() # (人数, 17, 3) -> x, y, conf
        boxes_data = r.boxes.xyxy.cpu().numpy()   # (人数, 4) -> バウンディングボックス
        
        if best_bat is not None:
            # バットが検出されている場合は、「バットに最も近い人」を打者とする（キャッチャーや審判を除外）
            bat_x1, bat_y1, bat_x2, bat_y2, _ = best_bat
            bat_cx = (bat_x1 + bat_x2) / 2
            bat_cy = (bat_y1 + bat_y2) / 2
            
            min_dist = float('inf')
            best_person_idx = -1
            
            for p_idx, p_box in enumerate(boxes_data):
                px1, py1, px2, py2 = p_box
                pcx = (px1 + px2) / 2
                pcy = (py1 + py2) / 2
                
                # バットの中心と人物の中心の距離を計算
                dist = np.sqrt((bat_cx - pcx)**2 + (bat_cy - pcy)**2)
                if dist < min_dist:
                    min_dist = dist
                    best_person_idx = p_idx
                    
            if best_person_idx != -1:
                target_person_kps = kps_data[best_person_idx]
                target_person_box = boxes_data[best_person_idx]
                
        else:
            # 万が一バットが未検出の場合は、とりあえず一番大きく映っている人（面積最大）を打者とみなす
            max_area = 0
            best_person_idx = -1
            for p_idx, p_box in enumerate(boxes_data):
                px1, py1, px2, py2 = p_box
                area = (px2 - px1) * (py2 - py1)
                if area > max_area:
                    max_area = area
                    best_person_idx = p_idx
                    
            if best_person_idx != -1:
                target_person_kps = kps_data[best_person_idx]
                target_person_box = boxes_data[best_person_idx]
                
    # ----------------------------------------
    # 3. CSV用のデータ作成
    # ----------------------------------------
    row_data = {
        "filename": filename,
        "frame_index": frame_idx,
        "bat_detected": 1 if best_bat is not None else 0,
        "bat_x1": round(best_bat[0], 2) if best_bat else "",
        "bat_y1": round(best_bat[1], 2) if best_bat else "",
        "bat_x2": round(best_bat[2], 2) if best_bat else "",
        "bat_y2": round(best_bat[3], 2) if best_bat else "",
        "bat_conf": round(best_bat[4], 4) if best_bat else "",
        "person_detected": 1 if target_person_kps is not None else 0
    }
    
    # 17点のキーポイント (x, y, confidence) を追加
    if target_person_kps is not None:
        for i, kp_name in enumerate(KEYPOINT_NAMES):
            x, y, conf = target_person_kps[i]
            row_data[f"{kp_name}_x"] = round(float(x), 2)
            row_data[f"{kp_name}_y"] = round(float(y), 2)
            row_data[f"{kp_name}_conf"] = round(float(conf), 4)
    else:
        for kp_name in KEYPOINT_NAMES:
            row_data[f"{kp_name}_x"] = ""
            row_data[f"{kp_name}_y"] = ""
            row_data[f"{kp_name}_conf"] = ""
            
    all_rows.append(row_data)
    
    # ----------------------------------------
    # 4. 可視化（画像保存）
    # ----------------------------------------
    annotated_img = img.copy()
    
    # バットの描画 (緑色枠)
    if best_bat is not None:
        bx1, by1, bx2, by2, bconf = best_bat
        cv2.rectangle(annotated_img, (int(bx1), int(by1)), (int(bx2), int(by2)), (0, 255, 0), 2)
        cv2.putText(annotated_img, f"Bat {bconf:.2f}", (int(bx1), int(by1)-10), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0,255,0), 2)
        
    # 打者の描画 (青色枠、赤色点)
    if target_person_kps is not None:
        px1, py1, px2, py2 = target_person_box
        cv2.rectangle(annotated_img, (int(px1), int(py1)), (int(px2), int(py2)), (255, 0, 0), 2)
        cv2.putText(annotated_img, "Batter", (int(px1), int(py1)-10), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255,0,0), 2)
        
        for i, (x, y, conf) in enumerate(target_person_kps):
            if conf > 0.5: # 信頼度が50%以上の点だけ描画
                cv2.circle(annotated_img, (int(x), int(y)), 4, (0, 0, 255), -1)
                
    out_path = os.path.join(OUTPUT_DIR, filename)
    cv2.imwrite(out_path, annotated_img)

# ============================================
# --- 最終結果のCSV保存 ---
# ============================================
if all_rows:
    df = pd.DataFrame(all_rows)
    df.to_csv(OUTPUT_CSV, index=False)
    print("=" * 50)
    print(f"✅ 完了！ CSVを保存しました: {OUTPUT_CSV}")
    print(f"✅ 描画済み画像を {OUTPUT_DIR}/ に保存しました。")
    print("=" * 50)
else:
    print("\n⚠️ 処理できる画像がありませんでした。")
```

## ステップ 4：結果の確認

実行が完了すると、Colab左側のファイルツリーに以下の2つが生成されます。
1. **`batting_2d_tracking.csv`**
   - これをダウンロードしてExcel等で開くと、各フレームごとのバット座標と全身17点の座標（X, Y, 信頼度）が時系列で記録されています。
2. **`output_frames` フォルダ**
   - 中に入っている画像をダウンロードして開くと、AIが打者とバットを正しく認識し、赤い点で骨格を描画できているか（キャッチャーを誤認識していないか）を視覚的に確認できます。
